// Google Apps Script for U.S. Industrial Suppliers website enquiries
// 1. Create a Google Sheet with a tab named Leads.
// 2. Extensions > Apps Script. Replace the editor contents with this code.
// 3. Deploy > New deployment > Web app; Execute as: Me; Who has access: Anyone.
// 4. Copy the Web app URL into index.html as SHEETS_ENDPOINT.
const SHEET_NAME = 'Leads';
function doPost(e) {
  const data = JSON.parse(e.postData.contents || '{}');
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Submitted at','Name','Company','Email','Phone','Product','Requirements']);
  }
  sheet.appendRow([
    new Date(), data.name || '', data.company || '', data.email || '',
    data.phone || '', data.product || '', data.message || ''
  ]);
  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
